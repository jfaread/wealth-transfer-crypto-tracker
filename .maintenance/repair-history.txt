async function repairChartHistory(bundle,wanted,signal){
 const ageOf=d=>d.historySavedAt||bundle.spec.historySavedAt||bundle.savedAt;
 const existing=new Map(bundle.data),ttl=bundle.spec.step===86400000?43200000:1800000;
 const pending=wanted.filter(id=>Date.now()-ageOf(existing.get(id))>ttl);
 if(!pending.length)return bundle;
 if(signal.aborted)throw new DOMException('Cancelled','AbortError');
 const historyController=new AbortController(),cancelHistory=()=>historyController.abort();signal.addEventListener('abort',cancelHistory,{once:true});
 const timer=setTimeout(cancelHistory,15000),historySignal=historyController.signal;
 const spec={...bundle.spec,end:Math.max(bundle.spec.end,Date.now()),saved:false};
 const repaired=new Map();let cursor=0;
 async function work(){
  while(cursor<pending.length&&!historySignal.aborted){const id=pending[cursor++];try{const value=await fetchAggregate(id,spec,historySignal,true);if(!historySignal.aborted)repaired.set(id,value);}catch(e){if(e.status===429)break;}}
 }
 try{
  // Two bounded lanes avoid a slow coin blocking all others. Every successful
  // result survives another coin's failure or the shared repair deadline.
  await Promise.all([work(),work()]);
  if(signal.aborted)throw new DOMException('Cancelled','AbortError');
  if(!repaired.size)return bundle;
  const data=bundle.data.map(([id,d])=>[id,repaired.get(id)||{...d,historySavedAt:ageOf(d)}]);
  const checkedAt=Math.min(...data.map(([,d])=>d.historySavedAt)),now=Date.now();
  const next={spec:{...spec,historySavedAt:checkedAt},data,savedAt:now};
  if(!validBundle(next))return bundle;
  return next;
 }finally{clearTimeout(timer);signal.removeEventListener('abort',cancelHistory);}
}
